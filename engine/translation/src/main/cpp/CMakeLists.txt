cmake_minimum_required(VERSION 3.22.1)
project(subloka_translation LANGUAGES C CXX)
include(FetchContent)
add_compile_options(-O3)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER GGML_OPENMP GGML_NATIVE GGML_CPU_ALL_VARIANTS)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
if(ANDROID_ABI STREQUAL "x86_64")
    foreach(option GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C)
        set(${option} OFF CACHE BOOL "" FORCE)
    endforeach()
endif()
FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
    GIT_TAG 2a667032dc788c3a071d538e0b572e45a3f12731)
FetchContent_MakeAvailable(llama)
add_library(subloka_translation SHARED gemma_jni.cpp)
target_compile_features(subloka_translation PRIVATE cxx_std_17)
# Whisper uses a different ggml revision. Prevent dynamic symbol interposition.
target_link_options(subloka_translation PRIVATE "-Wl,--exclude-libs,ALL")
target_link_libraries(subloka_translation PRIVATE llama android log)
